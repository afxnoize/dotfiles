# B2 CLI deny/ask patterns for validate-b2.sh
# Format: <regex> | <decision> | <reason>
# decision: deny = block, ask = prompt user
# Lines starting with # are comments, blank lines are ignored.

# ── rm (delete files) ──
b2\s+rm\s+.*--recursive            | deny | Recursive B2 file deletion is blocked.
b2\s+rm\s+.*--versions             | deny | Deleting all file versions is blocked.
b2\s+rm\s+                         | ask  | Removing B2 files. Proceed?

# ── bucket ──
b2\s+bucket\s+delete                | deny | B2 bucket deletion is blocked.
b2\s+bucket\s+update                | ask  | Updating B2 bucket settings. Proceed?
b2\s+bucket\s+create                | ask  | Creating B2 bucket. Proceed?

# ── key (credentials) ──
b2\s+key\s+delete                   | deny | Deleting B2 application key is blocked.
b2\s+key\s+create                   | deny | Creating B2 application key is blocked.

# ── account ──
b2\s+account\s+authorize            | ask  | Changing B2 authorization. Proceed?

# ── sync (destructive flags) ──
b2\s+sync\s+.*--delete              | ask  | B2 sync with --delete flag. Proceed?
b2\s+sync\s+.*--replace-newer       | ask  | B2 sync replacing newer files. Proceed?

# ── large file management ──
b2\s+file\s+large\s+unfinished\s+cancel | ask | Cancelling unfinished large files. Proceed?

# ── replication ──
b2\s+replication\s+delete           | ask  | Deleting replication rule. Proceed?

# ── legacy commands (still functional) ──
b2\s+delete-bucket                  | deny | B2 bucket deletion is blocked.
b2\s+delete-file-version            | ask  | Deleting B2 file version. Proceed?
b2\s+cancel-all-unfinished-large-files | ask | Cancelling all unfinished large files. Proceed?
